Sample Queries — Run These in MySQL Workbench

Copy-paste ready. These prove the data is live and the constraints work.

-- 1. All tables in the database
SHOW TABLES;

-- 2. Row counts per core table
SELECT 'users' AS tbl, COUNT(*) FROM users
UNION ALL SELECT 'resources', COUNT(*) FROM resources
UNION ALL SELECT 'bookings', COUNT(*) FROM bookings
UNION ALL SELECT 'notifications', COUNT(*) FROM notifications;

-- 3. All resources with their building and category
SELECT r.code, r.name, r.capacity, r.status,
       c.name AS category, b.name AS building
FROM resources r
JOIN resource_categories c ON c.id = r.category_id
JOIN rooms rm ON rm.id = r.room_id
JOIN buildings b ON b.id = rm.building_id
ORDER BY r.code;

-- 4. Pending bookings waiting for admin approval
SELECT booking_code, purpose, start_time, status
FROM bookings WHERE status = 'pending' LIMIT 10;

-- 5. Double-booking protection (this INSERT must FAIL)
INSERT INTO bookings (booking_code, resource_id, user_id,
  start_time, end_time, purpose, attendees)
VALUES ('TEST-OVERLAP', 1, 1,
  '2026-11-27 10:00:00', '2026-11-27 12:00:00',
  'Overlap test — should be rejected', 5);
-- Expected: trigger rejects it with "That slot was just taken"
